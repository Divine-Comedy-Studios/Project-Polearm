\section{Schlussfolgerungen} \label{sec:Schlussfolgerungen}

\subsection{Beantwortung der Erkenntnisfrage}

\begin{itemize}[leftmargin=1.5cm]
    \item \textcolor{red}{Erkenntnisfrage wörtlich aufgreifen und explizit beantworten, auf Grundlage von Recherche, Analyse, Prototyp und Playtest (Vertrag C.5).}
    \item \textcolor{red}{Gliederung entlang der zwei Komponenten (Gegnerverhalten, Gegnerplatzierung) und der drei Qualitäten (Spielfluss, Schwierigkeit, Spielgefühl).}
    \item \textcolor{red}{Am Schluss den Bogen zum Einstieg in Kapitel~1 schliessen.}
\end{itemize}

\subsection{Wichtigste Erkenntnisse}

\begin{itemize}[leftmargin=1.5cm]
    \item \textcolor{red}{3--5 Kernaussagen, inhaltlich und aus dem Arbeitsprozess; keine Details aus Kapitel~\ref{sec:Evaluation} wiederholen.}
\end{itemize}

\subsection{Limitationen}

\begin{itemize}[leftmargin=1.5cm]
    \item \textcolor{red}{Kleine Stichprobe, Testpersonen aus dem Umfeld, kein Tutorial, Analysen nur von aussen rekonstruiert, Quellen nicht genrespezifisch, fehlende Zeitstempel, Zeitmangel bei der Auswahl der Recherchepunkte.}
\end{itemize}

\subsection{Ausblick}

\begin{itemize}[leftmargin=1.5cm]
    \item \textcolor{red}{2--4 Punkte, möglichst aus den Daten abgeleitet (z.\,B. Spawn-Art als Variable, Höhen und Flanken, akustische Angriffsankündigung).}
\end{itemize}
